cmake_minimum_required(VERSION 3.16)
project(memshift VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(memshiftd src/memshiftd.cpp src/harmony_v1.cpp src/device_oracle_evidence.cpp)
target_link_libraries(memshiftd PRIVATE Threads::Threads)
add_executable(memshiftctl src/memshiftctl.cpp)
add_executable(compare_runs tools/compare_runs.cpp)
# Linux OOM lab workload (linux_lab/); not used by the Harmony backends.
add_executable(memlab_app linux_lab/memlab_app.cpp)
target_link_libraries(memlab_app PRIVATE Threads::Threads)
add_executable(test_policy tests/test_policy.cpp)
add_executable(test_harmony_v1 tests/test_harmony_v1.cpp src/harmony_v1.cpp src/device_oracle_evidence.cpp)
target_link_libraries(test_harmony_v1 PRIVATE Threads::Threads)
add_executable(harmony_v1_probe tools/harmony_v1_probe.cpp)
target_link_libraries(harmony_v1_probe PRIVATE Threads::Threads)
add_executable(harmony_v1_isolated_workload examples/harmony_v1_isolated_workload.cpp)
add_executable(harmony_v1_test_reclaim tools/harmony_v1_test_reclaim.cpp)
enable_testing()
add_test(NAME policy COMMAND test_policy)
add_executable(test_linux_scope tests/test_linux_scope.cpp)
add_test(NAME linux_scope COMMAND test_linux_scope)
add_executable(test_watermark tests/test_watermark.cpp)
add_test(NAME watermark COMMAND test_watermark)
add_test(NAME harmony_v1_profile COMMAND test_harmony_v1)
install(TARGETS memshiftd memshiftctl compare_runs harmony_v1_probe harmony_v1_isolated_workload harmony_v1_test_reclaim RUNTIME DESTINATION bin)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py ${CMAKE_CURRENT_BINARY_DIR})
endif()
